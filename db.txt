Table users {
  id uuid [pk]
  email varchar(255) [not null, unique]
  password_hash varchar(255) [not null]
  full_name varchar(150) [not null]
  platform_role platform_role [not null, default: 'USER']
  is_active boolean [not null, default: true]
  last_login_at timestamptz
  created_at timestamptz [not null]
  updated_at timestamptz [not null]
}

Table tenants {
  id uuid [pk]
  name varchar(200) [not null]
  legal_name varchar(255)
  slug varchar(120) [not null, unique]
  tax_number varchar(50)
  country_code varchar(2)
  is_active boolean [not null, default: true]
  created_at timestamptz [not null]
  updated_at timestamptz [not null]
}

Table tenant_memberships {
  id uuid [pk]
  tenant_id uuid [not null, ref: > tenants.id]
  user_id uuid [not null, ref: > users.id]
  role tenant_role [not null]
  is_active boolean [not null, default: true]
  created_at timestamptz [not null]
  updated_at timestamptz [not null]

  indexes {
    (tenant_id, user_id) [unique]
    user_id
  }
}

Table facilities {
  id uuid [pk]
  tenant_id uuid [not null, ref: > tenants.id]
  name varchar(200) [not null]
  code varchar(50)
  country_code varchar(2)
  city varchar(120)
  district varchar(120)
  address text
  is_active boolean [not null, default: true]
  created_at timestamptz [not null]
  updated_at timestamptz [not null]

  indexes {
    (tenant_id, code) [unique]
  }
}

Table carbon_assessments {

  updated_at timestamptz [not null]

  indexes {
    tenant_id
    (tenant_id, period_start, period_end)
  }
}

Table assessment_facilities {

  facility_id uuid [not null, ref: > facilities.id]
  created_at timestamptz [not null]

  indexes {

  }

Table units {
  id uuid [pk]
  code varchar(30) [not null, unique]
  name varchar(100) [not null]
  symbol varchar(30) [not null]

  is_active boolean [not null, default: true]

Table natural_gas_activities {
  id uuid [pk]
  tenant_id uuid [not null, ref: > tenants.id]
  assessment_id uuid [not null, ref: > carbon_assessments.id]
  facility_id uuid [not null, ref: > facilities.id]
  period_start date [not null]
  period_end date [not null]
  consumption_value decimal(20,6) [not null]
  unit_id uuid [not null, ref: > units.id]

  supplier_name varchar(200)

  notes text

  validated_by_user_id uuid [ref: > users.id]
  validated_at timestamptz
  created_at timestamptz [not null]
  updated_at timestamptz [not null]

  indexes {
    tenant_id
    assessment_id
    facility_id
    (tenant_id, facility_id, period_start, period_end)
  }
}
